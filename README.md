# Kompilasi tugas kampus unand tercinta


## Compile syntax (for c)
g++ -std=c++17 -Wshadow -Wall -o "%e" "%f" -O2 -Wno-unused-result

atau

g++ -std=c++17 -Wshadow -Wall -o "%e" "%f" -g -fsanitize=address -fsanitize=undefined -D_GLIBCXX_DEBUG
